# Country scope, after Nat_Main: the revenue lines the census does not sum nationally by themselves.
# MTUI_Special: income that is added to revenue outside the provinces. MTUI_OtherIndirect: what is left of
# indirect taxation after the lines the budget tooltips list (commerce taxes, tribal tributes, trade post fees...).
mtui_budget_extra = {
	set_key = { lhs = MTUI_Special which = TaxD_Teuton }
	change_key = { lhs = MTUI_Special which = TaxD_HRE }
	change_key = { lhs = MTUI_Special which = TaxD_Knights }
	change_key = { lhs = MTUI_Special which = Post_Fees }
	set_key = { lhs = MTUI_OtherIndirect which = TaxD_Indirect }
	subtract_key = { lhs = MTUI_OtherIndirect which = TaxD_Fees }
	subtract_key = { lhs = MTUI_OtherIndirect which = Tariff_Income }
	subtract_key = { lhs = MTUI_OtherIndirect which = TaxD_ExciseBU }
	subtract_key = { lhs = MTUI_OtherIndirect which = TaxD_Industry }
	subtract_key = { lhs = MTUI_OtherIndirect which = TaxD_Obligations }
}

# Round variable $var$ to 0 or 1 decimals. EU4 variables are fixed point with 3 decimals, so
# x * 0.001 * 1000 drops the fraction (the whole_part trick); +-0.5 first makes it round.
mtui_round0 = {
	if = {
		limit = { check_variable = { which = $var$ value = 0 } }
		change_variable = { which = $var$ value = 0.5 }
	}
	else = {
		subtract_variable = { which = $var$ value = 0.5 }
	}
	multiply_variable = { which = $var$ value = 0.001 }
	multiply_variable = { which = $var$ value = 1000 }
}
mtui_round1 = {
	multiply_variable = { which = $var$ value = 10 }
	mtui_round0 = { var = $var$ }
	divide_variable = { which = $var$ value = 10 }
}

# open a tab with fresh values (top bar shortcuts)
mtui_atlas_open_refresh = {
	mtui_atlas_open_tab = { tab = $tab$ }
	# refresh through a hidden event: the game evaluates the conditions inside a button's effect (province loops)
	# from the country to build its tooltip, which logs scope errors; it does not look into events
	hidden_effect = { country_event = { id = mtui_ui.2 } }
}
